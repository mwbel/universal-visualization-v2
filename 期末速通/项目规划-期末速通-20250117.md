// 考点追踪组件
class CheckListTracker {
  constructor(course) {
    this.course = course;
    this.checkList = checkLists[course];
  }

  markMastered(topicId) {
    const topic = this.checkList.find(t => t.id === topicId);
    if (topic) {
      topic.mastered = true;
      this.saveProgress();
    }
  }

  getProgress() {
    const total = this.checkList.length;
    const mastered = this.checkList.filter(t => t.mastered).length;
    return {
      percentage: (mastered / total) * 100,
      mastered: mastered,
      total: total,
      remaining: total - mastered
    };
  }

  getPriorityTopics() {
    return this.checkList
      .filter(t => !t.mastered && t.priority === 'critical')
      .map(t => t.topic);
  }
}
