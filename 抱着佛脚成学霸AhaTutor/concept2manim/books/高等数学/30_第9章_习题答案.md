1. (1)  $\int_0^1\mathrm{d}x\int_0^x f(x,y)\mathrm{d}y + \int_1^2\mathrm{d}x\int_0^{2 - x}f(x,y)\mathrm{d}y = \int_0^1\mathrm{d}y\int_y^{2 - y}f(x,y)\mathrm{d}x;$

(2)  $\int_{-1}^{1}\mathrm{d}x\int_{x^2}^{1}f(x,y)\mathrm{d}y = \int_0^1\mathrm{d}y\int_{-\sqrt{y}}^{\sqrt{y}}f(x,y)\mathrm{d}x;$

(3)  $\int_0^1\mathrm{d}x\int_{\frac{x}{2}}^{2x}f(x,y)\mathrm{d}y + \int_1^2\mathrm{d}x\int_{\frac{x}{2}}^{\frac{2}{x}}f(x,y)\mathrm{d}y$

$$
= \int_ {0} ^ {1} \mathrm {d} y \int_ {\frac {y}{2}} ^ {2 y} f (x, y) \mathrm {d} x + \int_ {1} ^ {2} \mathrm {d} y \int_ {\frac {y}{2}} ^ {\frac {2}{y}} f (x, y) \mathrm {d} x;
$$

(4)  $\int_{-a}^{a}\mathrm{d}x\int_{a - \sqrt{a^2 - x^2}}^{a + \sqrt{a^2 - x^2}}f(x,y)\mathrm{d}y = \int_0^{2a}\mathrm{d}y\int_{-\sqrt{2ay - y^2}}^{\sqrt{2ay - y^2}}f(x,y)\mathrm{d}x;$

(5)  $\int_{3}^{5}\mathrm{d}x\int_{\frac{3x + 1}{2}}^{\frac{3x + 4}{2}}f(x,y)\mathrm{d}y$

$$
= \int_ {5} ^ {\frac {1 3}{2}} \mathrm {d} y \int_ {3} ^ {\frac {2 y - 1}{3}} f (x, y) \mathrm {d} x + \int_ {\frac {1 3}{2}} ^ {8} \mathrm {d} y \int_ {\frac {2 y - 4}{3}} ^ {\frac {2 y - 1}{3}} f (x, y) \mathrm {d} x + \int_ {8} ^ {\frac {1 9}{2}} \mathrm {d} y \int_ {\frac {2 y - 4}{3}} ^ {5} f (x, y) \mathrm {d} x.
$$
