## 🎯 为什么选择本地数据库

### 本地数据库 vs 云数据库

| 维度 | 本地数据库 | 云数据库 |
|------|-----------|---------|
| **成本** | ✅ 免费 | ❌ 按使用量付费 |
| **性能** | ✅ 零延迟 | ⚠️ 网络延迟 |
| **数据安全** | ✅ 完全掌控 | ⚠️ 依赖第三方 |
| **开发调试** | ✅ 方便 | ⚠️ 需要网络 |
| **部署运维** | ⚠️ 需要自己维护 | ✅ 托管服务 |
| **扩展性** | ⚠️ 受限于硬件 | ✅ 弹性扩展 |

**推荐使用场景**：
- ✅ 开发和测试阶段
- ✅ 个人项目或小团队
- ✅ 数据量 < 100GB
- ✅ 预算有限的项目

## 📦 PostgreSQL 安装

### macOS 安装（推荐）

#### 方法1：使用 Homebrew（最简单）

```bash
# 1. 安装 Homebrew（如果未安装）
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"

# 2. 安装 PostgreSQL
brew install postgresql@16

# 3. 启动 PostgreSQL 服务
brew services start postgresql@16

# 4. 验证安装
psql --version
# 输出: psql (PostgreSQL) 16.x.x
```

#### 方法2：使用 PostgreSQL Installer

```bash
# 1. 下载安装包
# 访问: https://www.postgresql.org/download/macosx/

# 2. 安装后添加到 PATH
export PATH=/Applications/PostgreSQL/16/bin:$PATH

# 3. 验证
psql --version
```

### 步骤2：创建数据库表结构

```bash
# 在项目根目录执行
cd /Users/Min369/Documents/同步空间/Manju/Projects/AlVisualization

# 创建数据库脚本目录
mkdir -p backend-v2/database

# 创建表结构 SQL 文件
cat > backend-v2/database/schema.sql << 'EOF'
-- ============================================
-- 万物可视化项目 - 数据库表结构
-- ============================================

-- 1. 学科表
CREATE TABLE IF NOT EXISTS subjects (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    name_en VARCHAR(100),
    description TEXT,
    icon VARCHAR(50),
    color VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. 学科关键词表
CREATE TABLE IF NOT EXISTS subject_keywords (
    id SERIAL PRIMARY KEY,
    subject_id INTEGER REFERENCES subjects(id) ON DELETE CASCADE,
    keyword VARCHAR(200) NOT NULL,
    weight FLOAT DEFAULT 1.0,
    is_core BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(subject_id, keyword)
);

-- 3. 关键词组合规则表
CREATE TABLE IF NOT EXISTS keyword_rules (
    id SERIAL PRIMARY KEY,
    subject_id INTEGER REFERENCES subjects(id) ON DELETE CASCADE,
    rule_name VARCHAR(100) NOT NULL,
    keywords TEXT[] NOT NULL,
    match_type VARCHAR(20) DEFAULT 'AND',  -- 'AND' 或 'OR'
    weight FLOAT DEFAULT 1.5,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. 核心概念词库表
CREATE TABLE IF NOT EXISTS core_concepts (
    id SERIAL PRIMARY KEY,
    subject_id INTEGER REFERENCES subjects(id) ON DELETE CASCADE,
    concept_name VARCHAR(200) NOT NULL,
    concept_name_en VARCHAR(200),
    description TEXT,
    difficulty_level VARCHAR(20),  -- 'basic', 'intermediate', 'advanced'
    prerequisite_concepts TEXT[],  -- 前置概念ID数组
    related_concepts TEXT[],       -- 相关概念ID数组
    visualization_type VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(subject_id, concept_name)
);

-- 5. 可视化模板表
CREATE TABLE IF NOT EXISTS visualization_templates (
    id SERIAL PRIMARY KEY,
    template_name VARCHAR(100) NOT NULL UNIQUE,
    subject_id INTEGER REFERENCES subjects(id) ON DELETE CASCADE,
    template_type VARCHAR(50) NOT NULL,  -- 'manim', 'plotly', 'd3', 'threejs', etc.
    description TEXT,
    difficulty_level VARCHAR(20),

    -- 模板配置
    template_code TEXT NOT NULL,
    config_schema JSONB,
    example_params JSONB,

    -- 适用概念
    applicable_concepts TEXT[],

    -- 元数据
    preview_image VARCHAR(500),
    tags TEXT[],
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 6. 文档-模板映射表
CREATE TABLE IF NOT EXISTS document_template_mappings (
    id SERIAL PRIMARY KEY,
    doc_id VARCHAR(100) NOT NULL,
    subject_id INTEGER REFERENCES subjects(id) ON DELETE CASCADE,
    detected_concepts TEXT[],
    recommended_template_id INTEGER REFERENCES visualization_templates(id),
    confidence_score FLOAT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    metadata JSONB
);

-- 创建索引
CREATE INDEX idx_subject_keywords_subject_id ON subject_keywords(subject_id);
CREATE INDEX idx_subject_keywords_keyword ON subject_keywords(keyword);
CREATE INDEX idx_core_concepts_subject_id ON core_concepts(subject_id);
CREATE INDEX idx_core_concepts_name ON core_concepts(concept_name);
CREATE INDEX idx_visualization_templates_subject_id ON visualization_templates(subject_id);
CREATE INDEX idx_visualization_templates_type ON visualization_templates(template_type);
CREATE INDEX idx_document_mappings_doc_id ON document_template_mappings(doc_id);

-- 创建视图：学科关键词汇总
CREATE OR REPLACE VIEW v_subject_keywords_summary AS
SELECT
    s.id AS subject_id,
    s.name AS subject_name,
    COUNT(sk.id) AS total_keywords,
    COUNT(sk.id) FILTER (WHERE sk.is_core = TRUE) AS core_keywords_count
FROM subjects s
LEFT JOIN subject_keywords sk ON s.id = sk.subject_id
GROUP BY s.id, s.name;

-- 创建触发器：自动更新 updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_subjects_updated_at BEFORE UPDATE ON subjects
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_visualization_templates_updated_at BEFORE UPDATE ON visualization_templates
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

EOF

# 执行 SQL 文件
psql -U universal_viz_user -d universal_viz_db -f backend-v2/database/schema.sql

# 输入密码后，会创建所有表
```

### 步骤2：创建数据库配置文件

```bash
# 创建数据库配置
cat > backend-v2/config/database.py << 'EOF'
"""
数据库配置
"""
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
import os

# 数据库连接配置
DATABASE_CONFIG = {
    "host": os.getenv("DB_HOST", "localhost"),
    "port": int(os.getenv("DB_PORT", "5432")),
    "database": os.getenv("DB_NAME", "universal_viz_db"),
    "user": os.getenv("DB_USER", "universal_viz_user"),
    "password": os.getenv("DB_PASSWORD", "your_secure_password_here"),
}

# 同步数据库 URL
SYNC_DATABASE_URL = f"postgresql://{DATABASE_CONFIG['user']}:{DATABASE_CONFIG['password']}@{DATABASE_CONFIG['host']}:{DATABASE_CONFIG['port']}/{DATABASE_CONFIG['database']}"

# 异步数据库 URL
ASYNC_DATABASE_URL = f"postgresql+asyncpg://{DATABASE_CONFIG['user']}}:{DATABASE_CONFIG['password']}@{DATABASE_CONFIG['host']}:{DATABASE_CONFIG['port']}/{DATABASE_CONFIG['database']}"

# 创建同步引擎
sync_engine = create_engine(
    SYNC_DATABASE_URL,
    pool_pre_ping=True,
    pool_size=5,
    max_overflow=10
)

# 创建异步引擎
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
async_engine = create_async_engine(
    ASYNC_DATABASE_URL,
    pool_pre_ping=True,
    pool_size=5,
    max_overflow=10
)

# 创建会话工厂
SyncSessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=sync_engine
)

AsyncSessionLocal = sessionmaker(
    async_engine,
    class_=AsyncSession,
    expire_on_commit=False
)

# 基类
Base = declarative_base()

# 依赖项：获取数据库会话
from fastapi import Depends
from typing import Generator

def get_db() -> Generator:
    """
    获取同步数据库会话
    """
    db = SyncSessionLocal()
    try:
        yield db
    finally:
        db.close()

async def get_async_db() -> AsyncSession:
    """
    获取异步数据库会话
    """
    async with AsyncSessionLocal() as session:
        yield session

# 测试连接
def test_connection():
    """
    测试数据库连接
    """
    try:
        conn = sync_engine.connect()
        conn.execute("SELECT 1")
        conn.close()
        print("✅ 数据库连接成功！")
        return True
    except Exception as e:
        print(f"❌ 数据库连接失败: {str(e)}")
        return False

if __name__ == "__main__":
    test_connection()

EOF
```

### 步骤3：创建 ORM 模型

```bash
# 创建 ORM 模型文件
cat > backend-v2/models/database_models.py << 'EOF'
"""
数据库 ORM 模型
"""
from sqlalchemy import Column, Integer, String, Text, Float, Boolean, DateTime, ForeignKey, ARRAY
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func
from config.database import Base

class Subject(Base):
    """学科表"""
    __tablename__ = "subjects"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(100), unique=True, nullable=False)
    name_en = Column(String(100))
    description = Column(Text)
    icon = Column(String(50))
    color = Column(String(20))
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), onupdate=func.now())

    # 关系
    keywords = relationship("SubjectKeyword", back_populates="subject", cascade="all, delete-orphan")
    concepts = relationship("CoreConcept", back_populates="subject", cascade="all, delete-orphan")
    templates = relationship("VisualizationTemplate", back_populates="subject", cascade="all, delete-orphan")

class SubjectKeyword(Base):
    """学科关键词表"""
    __tablename__ = "subject_keywords"

    id = Column(Integer, primary_key=True, index=True)
    subject_id = Column(Integer, ForeignKey("subjects.id", ondelete="CASCADE"), nullable=False)
    keyword = Column(String(200), nullable=False)
    weight = Column(Float, default=1.0)
    is_core = Column(Boolean, default=False)
    created_at = Column(DateTime(timezone=True), server_default=func.now())

    # 关系
    subject = relationship("Subject", back_populates="keywords")

class CoreConcept(Base):
    """核心概念词库表"""
    __tablename__ = "core_concepts"

    id = Column(Integer, primary_key=True, index=True)
    subject_id = Column(Integer, ForeignKey("subjects.id", ondelete="CASCADE"), nullable=False)
    concept_name = Column(String(200), nullable=False)
    concept_name_en = Column(String(200))
    description = Column(Text)
    difficulty_level = Column(String(20))
    prerequisite_concepts = Column(ARRAY(Integer))
    related_concepts = Column(ARRAY(Integer))
    visualization_type = Column(String(50))
    created_at = Column(DateTime(timezone=True), server_default=func.now())

    # 关系
    subject = relationship("Subject", back_populates="concepts")

class VisualizationTemplate(Base):
    """可视化模板表"""
    __tablename__ = "visualization_templates"

    id = Column(Integer, primary_key=True, index=True)
    template_name = Column(String(100), unique=True, nullable=False)
    subject_id = Column(Integer, ForeignKey("subjects.id", ondelete="CASCADE"))
    template_type = Column(String(50), nullable=False)
    description = Column(Text)
    difficulty_level = Column(String(20))
    template_code = Column(Text, nullable=False)
    config_schema = Column(Text)  # JSONB
    example_params = Column(Text)  # JSONB
    applicable_concepts = Column(ARRAY(String))
    preview_image = Column(String(500))
    tags = Column(ARRAY(String))
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), onupdate=func.now())

    # 关系
    subject = relationship("Subject", back_populates="templates")

# 导出所有模型
__all__ = [
    "Subject",
    "SubjectKeyword",
    "CoreConcept",
    "VisualizationTemplate"
]

EOF
```

## ❓ 常见问题

### 问题1：连接被拒绝

```
Error: connection refused
```

**解决方案**：
```bash
# 检查 PostgreSQL 是否运行
brew services list  # macOS
sudo systemctl status postgresql  # Linux

# 启动服务
brew services start postgresql@16  # macOS
sudo systemctl start postgresql  # Linux
```

---

### 问题2：密码认证失败

```
Error: password authentication failed
```

**解决方案**：
```bash
# 重置用户密码
psql -U postgres -d postgres
ALTER USER universal_viz_user WITH PASSWORD 'new_password_here';
\q

# 更新 .env 文件中的 DB_PASSWORD
```

---

### 问题3：端口冲突

```
Error: port 5432 already in use
```

**解决方案**：
```bash
# 查看占用端口的进程
lsof -i :5432  # macOS/Linux
netstat -ano | findstr :5432  # Windows

# 更改配置使用其他端口
# 在 .env 中设置 DB_PORT=5433
```

---

### 问题4：权限不足

```
Error: permission denied for table subjects
```

**解决方案**：
```sql
-- 连接到数据库
psql -U postgres -d universal_viz_db

-- 重新授予权限
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO universal_viz_user;
GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public TO universal_viz_user;
\q
```
