## 方案设计

### 双路径处理策略

```
用户上传PDF
    ↓
┌─────────────────────────────────────┐
│ 格式检测（判断是否为扫描件）          │
└────────┬────────────────────────────┘
         ↓
    是扫描件？
    ↓
   ┌─────┴─────┐
   │           │
  YES         NO
   │           │
   ↓           ↓
MinerU API   Gemini API
   │           │
   └─────┬─────┘
         ↓
    统一输出格式
    （Markdown + 结构化数据）
         ↓
      存储到数据库
```

### 为什么这样设计？

| 对比维度 | Gemini API | MinerU API |
|---------|-----------|------------|
| **适用场景** | 可编辑PDF（含文本层）| 扫描件/纯图片PDF |
| **优势** | 多模态理解强，保持格式 | OCR精度高，数学公式识别强 |
| **公式支持** | 支持LaTeX提取 | 专门优化数学公式 |
| **成本** | 较低 | 较高 |
| **速度** | 快 | 较慢 |

**关键洞察：自动检测PDF类型，选择最优方案！**

### 主模块：DocumentPreprocessor

```python
# modules/document_preprocessor.py

import os
import base64
import fitz  # PyMuPDF
from typing import Dict, Any, List, Tuple
from datetime import datetime
import hashlib
import json

class DocumentPreprocessor:
    """文档预处理模块"""

    def __init__(self):
        # 初始化API客户端
        self.gemini_client = GeminiAPIClient()
        self.mineru_client = MinerUAPIClient()

        # 配置参数
        self.max_file_size = 50 * 1024 * 1024  # 50MB
        self.supported_formats = ['application/pdf']

    async def process(self, file_path: str, user_id: str) -> Dict[str, Any]:
        """
        处理上传的文档

        Args:
            file_path: PDF文件路径
            user_id: 用户ID

        Returns:
            处理结果字典
        """
        # 1. 生成文档ID
        doc_id = self._generate_doc_id(file_path)

        # 2. 验证文件
        self._validate_file(file_path)

        # 3. 检测是否为扫描件
        is_scanned = await self._detect_if_scanned(file_path)

        print(f"文档 {doc_id} 检测结果: {'扫描件' if is_scanned else '可编辑PDF'}")

        # 4. 根据类型选择处理方案
        if is_scanned:
            # 扫描件 → MinerU API
            result = await self._process_with_mineru(file_path, doc_id)
        else:
            # 可编辑PDF → Gemini API
            result = await self._process_with_gemini(file_path, doc_id)

        # 5. 统一输出格式
        standardized_result = self._standardize_output(result, is_scanned)

        # 6. 过滤敏感信息
        standardized_result = self._filter_sensitive_info(standardized_result)

        # 7. 计算文件MD5（用于去重）
        file_md5 = self._calculate_md5(file_path)
        standardized_result['file_md5'] = file_md5

        # 8. 提取文档元数据
        standardized_result['metadata'] = self._extract_metadata(file_path)

        return standardized_result

    def _generate_doc_id(self, file_path: str) -> str:
        """生成唯一文档ID"""
        timestamp = datetime.now().strftime('%Y%m%d%H%M%S')
        import random
        random_num = random.randint(1000, 9999)
        return f"DOC{timestamp}{random_num}"

    def _validate_file(self, file_path: str):
        """验证文件"""
        # 检查文件是否存在
        if not os.path.exists(file_path):
            raise FileNotFoundError(f"文件不存在: {file_path}")

        # 检查文件大小
        file_size = os.path.getsize(file_path)
        if file_size > self.max_file_size:
            raise ValueError(f"文件过大: {file_size / 1024 / 1024:.2f}MB，最大支持50MB")

        # 检查文件格式
        if not file_path.lower().endswith('.pdf'):
            raise ValueError("仅支持PDF格式文件")

    async def _detect_if_scanned(self, file_path: str) -> bool:
        """
        检测PDF是否为扫描件

        方法：
        1. 使用PyMuPDF检查是否有文本层
        2. 文本层字符数 < 总页数*100 → 判定为扫描件
        """
        import fitz  # PyMuPDF

        doc = fitz.open(file_path)

        total_text_length = 0
        total_pages = len(doc)

        for page_num, page in enumerate(doc):
            text = page.get_text()
            total_text_length += len(text)

            # 优化：前3页都是图片，大概率是扫描件
            if page_num < 3 and len(text) < 50:
                doc.close()
                return True

        doc.close()

        # 判断标准：平均每页文本 < 100字符
        avg_text_per_page = total_text_length / max(total_pages, 1)

        is_scanned = avg_text_per_page < 100

        print(f"平均每页文本长度: {avg_text_per_page:.0f} 字符")
        print(f"检测结果: {'扫描件' if is_scanned else '可编辑PDF'}")

        return is_scanned

    async def _process_with_gemini(self, file_path: str, doc_id: str) -> Dict[str, Any]:
        """
        使用 Gemini API 处理可编辑PDF

        优势：
        - 保持文档结构
        - 提取LaTeX公式
        - 识别表格
        - 多模态理解
        """
        print(f"[Gemini] 开始处理文档: {doc_id}")

        try:
            # 调用 Gemini API
            result = await self.gemini_client.process_pdf(file_path)

            print(f"[Gemini] 处理完成: 提取了 {len(result['sections'])} 个章节")

            return {
                'success': True,
                'source': 'gemini',
                'content': result['content'],  # Markdown格式
                'structure': result['structure'],  # 章节结构
                'formulas': result['formulas'],  # LaTeX公式列表
                'tables': result['tables'],  # 表格数据
                'images': result['images'],  # 图片引用
                'processing_time': result['processing_time']
            }

        except Exception as e:
            print(f"[Gemini] 处理失败: {str(e)}")
            # 降级：尝试使用 MinerU
            print(f"[Gemini] 降级到 MinerU API")
            return await self._process_with_mineru(file_path, doc_id)

    async def _process_with_mineru(self, file_path: str, doc_id: str) -> Dict[str, Any]:
        """
        使用 MinerU API 处理扫描件

        优势：
        - 高精度OCR
        - 数学公式识别（LaTeX）
        - 保留格式（Markdown）
        - 表格识别
        """
        print(f"[MinerU] 开始处理文档: {doc_id}")

        try:
            # 调用 MinerU API
            result = await self.mineru_client.process_pdf(file_path)

            print(f"[MinerU] 处理完成: OCR识别了 {len(result['content'])} 字符")

            return {
                'success': True,
                'source': 'mineru',
                'content': result['markdown'],  # Markdown格式
                'structure': result['structure'],  # 章节结构
                'formulas': result['formulas'],  # LaTeX公式列表
                'tables': result['tables'],  # 表格数据
                'images': result['images'],  # 图片引用
                'ocr_confidence': result['confidence'],  # OCR置信度
                'processing_time': result['processing_time']
            }

        except Exception as e:
            print(f"[MinerU] 处理失败: {str(e)}")
            raise Exception(f"文档处理失败: {str(e)}")

    def _standardize_output(self, result: Dict[str, Any], is_scanned: bool) -> Dict[str, Any]:
        """
        统一输出格式

        无论使用 Gemini 还是 MinerU，输出统一格式
        """
        return {
            'doc_id': result.get('doc_id'),
            'success': result['success'],
            'processing_source': result['source'],  # 'gemini' or 'mineru'
            'is_scanned': is_scanned,

            # 文本内容（Markdown格式）
            'content': result['content'] or result.get('markdown', ''),

            # 结构化数据
            'structure': {
                'sections': result['structure'],  # 章节列表
                'total_pages': self._count_pages(result),
                'has_formulas': len(result.get('formulas', [])) > 0,
                'has_tables': len(result.get('tables', [])) > 0,
                'has_images': len(result.get('images', [])) > 0
            },

            # 提取的元素
            'formulas': result.get('formulas', []),  # LaTeX公式列表
            'tables': result.get('tables', []),  # 表格列表
            'images': result.get('images', []),  # 图片列表

            # 处理信息
            'processing_time': result.get('processing_time', 0),
            'ocr_confidence': result.get('ocr_confidence', None),
            'status': 'preprocessing_completed'
        }

    def _filter_sensitive_info(self, result: Dict[str, Any]) -> Dict[str, Any]:
        """
        过滤敏感信息

        正则匹配：手机号、身份证号、邮箱等
        """
        import re

        content = result['content']

        # 手机号（中国大陆）
        content = re.sub(r'1[3-9]\d{9}', '***', content)

        # 身份证号
        content = re.sub(r'\d{17}[\dXx]', '***', content)

        # 邮箱
        content = re.sub(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b', '***@***.com', content)

        # 银行卡号
        content = re.sub(r'\d{16,19}', '***', content)

        result['content'] = content

        return result

    def _calculate_md5(self, file_path: str) -> str:
        """计算文件MD5"""
        hash_md5 = hashlib.md5()
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(4096), b""):
                hash_md5.update(chunk)
        return hash_md5.hexdigest()

    def _extract_metadata(self, file_path: str) -> Dict[str, Any]:
        """提取文档元数据"""
        import fitz

        doc = fitz.open(file_path)
        first_page = doc[0]

        metadata = {
            'filename': os.path.basename(file_path),
            'file_size': os.path.getsize(file_path),
            'total_pages': len(doc),
            'created_date': datetime.now().isoformat(),
            'page_size': (first_page.rect.width, first_page.rect.height)
        }

        # 尝试提取PDF元数据
        pdf_metadata = doc.metadata
        if pdf_metadata.get('title'):
            metadata['title'] = pdf_metadata['title']
        if pdf_metadata.get('author'):
            metadata['author'] = pdf_metadata['author']
        if pdf_metadata.get('subject'):
            metadata['subject'] = pdf_metadata['subject']

        doc.close()

        return metadata

    def _count_pages(self, result: Dict[str, Any]) -> int:
        """估算页数"""
        # 如果有明确的页数信息
        if 'total_pages' in result:
            return result['total_pages']

        # 否则根据章节估算
        sections = result.get('structure', [])
        if sections and len(sections) > 0:
            # 简单估算：每章平均10页
            return min(len(sections) * 10, 1000)

        return 0
```

### Gemini API客户端

```python
# api_clients/gemini_client.py

import google.generativeai as genai
import os
from typing import Dict, Any, List
import base64
import time

class GeminiAPIClient:
    """Gemini 1.5 Pro API 客户端"""

    def __init__(self):
        self.api_key = os.getenv("GEMINI_API_KEY")
        if not self.api_key:
            raise ValueError("GEMINI_API_KEY not set")

        genai.configure(api_key=self.api_key)
        self.model = genai.GenerativeModel('gemini-1.5-pro')

    async def process_pdf(self, file_path: str) -> Dict[str, Any]:
        """
        使用 Gemini 处理可编辑PDF

        Args:
            file_path: PDF文件路径

        Returns:
            包含content、structure、formulas等字段的字典
        """
        start_time = time.time()

        # 1. 上传文件到 Gemini
        file = self._upload_file(file_path)

        # 2. 构建提示词
        prompt = self._build_prompt()

        # 3. 调用 Gemini
        response = await self._generate_content(file, prompt)

        # 4. 解析响应
        result = self._parse_response(response)

        result['processing_time'] = time.time() - start_time

        return result

    def _upload_file(self, file_path: str):
        """上传文件到 Gemini"""
        file_name = os.path.basename(file_path)

        print(f"[Gemini] 上传文件: {file_name}")

        file = genai.upload_file(path=file_path, display_name=file_name)

        # 等待文件处理完成
        while file.state.name == "PROCESSING":
            import asyncio
            await asyncio.sleep(2)
            file = genai.get_file(file.name)

        if file.state.name == "FAILED":
            raise ValueError(f"文件上传失败: {file.state.name}")

        print(f"[Gemini] 文件上传完成: {file.uri}")
        return file

    def _build_prompt(self) -> str:
        """构建提示词"""
        return """
请详细分析这份PDF文档，并以JSON格式返回以下信息：

{
  "content": "文档的完整Markdown文本，保留章节结构、公式、表格",
  "structure": [
    {
      "level": 1,
      "title": "章节标题",
      "page_range": [起始页, 结束页]
    }
  ],
  "formulas": [
    {
      "latex": "公式LaTeX代码",
      "description": "公式含义",
      "page": 页码,
      "context": "公式所在位置的上下文"
    }
  ],
  "tables": [
    {
      "caption": "表格标题",
      "rows": 行数,
      "columns": 列数,
      "page": 页码,
      "data": "表格数据（Markdown表格格式）"
    }
  ],
  "images": [
    {
      "description": "图片描述",
      "page": 页码,
      "figure_type": "图表/示意图/照片"
    }
  ],
  "summary": {
    "title": "文档标题",
    "abstract": "文档摘要（200字以内）",
    "keywords": ["关键词1", "关键词2"],
    "main_topics": ["主要主题1", "主要主题2"]
  }
}

**重要要求**：
1. content 字段包含完整的Markdown格式文档
2. 公式使用 LaTeX 格式（如 $E=mc^2$ 或 $$\\int_0^1 f(x)dx$$）
3. 保留原文档的章节层级（使用 # ## ### 标记）
4. 表格转换为 Markdown 表格格式
5. 确保返回的是有效的 JSON 格式

请开始分析：
"""

    async def _generate_content(self, file, prompt: str) -> str:
        """生成内容"""
        import asyncio

        try:
            # Gemini 不支持异步，使用线程池
            loop = asyncio.get_event_loop()
            response = await loop.run_in_executor(
                None,
                lambda: self.model.generate_content([
                    file,
                    prompt
                ])
            )

            return response.text

        except Exception as e:
            print(f"[Gemini] API调用失败: {str(e)}")
            raise

    def _parse_response(self, response: str) -> Dict[str, Any]:
        """解析 Gemini 响应"""
        import json
        import re

        # 尝试直接解析
        try:
            return json.loads(response)
        except:
            pass

        # 尝试提取 JSON 代码块
        json_match = re.search(r'```json\n(.*?)\n```', response, re.DOTALL)
        if json_match:
            try:
                return json.loads(json_match.group(1))
            except:
                pass

        # 尝试提取 { ... }
        json_match = re.search(r'\{.*\}', response, re.DOTALL)
        if json_match:
            try:
                return json.loads(json_match.group(0))
            except:
                pass

        # 如果都失败，返回原始响应
        return {
            'raw_response': response,
            'parse_error': '无法解析为JSON'
        }
```

### MinerU API客户端

```python
# api_clients/mineru_client.py

import requests
import os
from typing import Dict, Any
import time
import json

class MinerUAPIClient:
    """MinerU API 客户端"""

    def __init__(self):
        self.api_key = os.getenv("MINERU_API_KEY")
        self.api_url = os.getenv("MINERU_API_URL", "https://api.mineru.cn/v1/extract")

        if not self.api_key:
            raise ValueError("MINERU_API_KEY not set")

        self.session = requests.Session()
        self.session.headers.update({
            'Authorization': f'Bearer {self.api_key}',
            'Content-Type': 'application/json'
        })

    async def process_pdf(self, file_path: str) -> Dict[str, Any]:
        """
        使用 MinerU 处理扫描件PDF

        Args:
            file_path: PDF文件路径

        Returns:
            包含markdown、formulas、structure等字段的字典
        """
        start_time = time.time()

        # 1. 上传文件
        upload_result = await self._upload_file(file_path)

        task_id = upload_result['task_id']

        # 2. 轮询任务状态
        result = await self._poll_task_status(task_id)

        # 3. 下载结果
        markdown_content = await self._download_result(result['result_url'])

        # 4. 解析Markdown内容
        parsed_result = self._parse_markdown(markdown_content)

        parsed_result['processing_time'] = time.time() - start_time
        parsed_result['confidence'] = result.get('confidence', None)

        return parsed_result

    async def _upload_file(self, file_path: str) -> Dict[str, Any]:
        """上传文件到 MinerU"""
        file_name = os.path.basename(file_path)
        file_size = os.path.getsize(file_path)

        print(f"[MinerU] 上传文件: {file_name} ({file_size / 1024 / 1024:.2f}MB)")

        # 读取文件
        with open(file_path, 'rb') as f:
            file_content = f.read()

        # 上传
        response = self.session.post(
            f"{self.api_url}/upload",
            files={'file': (file_name, file_content, 'application/pdf')},
            data={
                'precision': 'high',  # 高精度
                'preserve_format': True,  # 保留格式
                'extract_formulas': True  # 提取公式
            }
        )

        if response.status_code != 200:
            raise Exception(f"文件上传失败: {response.text}")

        result = response.json()

        print(f"[MinerU] 文件上传成功, 任务ID: {result['task_id']}")

        return result

    async def _poll_task_status(self, task_id: str, max_wait: int = 300) -> Dict[str, Any]:
        """轮询任务状态"""
        import asyncio

        print(f"[MinerU] 等待任务完成: {task_id}")

        start_time = time.time()

        while True:
            # 检查超时
            if time.time() - start_time > max_wait:
                raise TimeoutError(f"任务处理超时: {task_id}")

            # 查询状态
            response = self.session.get(f"{self.api_url}/status/{task_id}")

            if response.status_code != 200:
                raise Exception(f"查询任务状态失败: {response.text}")

            result = response.json()
            status = result['status']

            print(f"[MinerU] 任务状态: {status}")

            if status == 'completed':
                print(f"[MinerU] 任务完成: {task_id}")
                return result

            elif status == 'failed':
                raise Exception(f"任务处理失败: {result.get('error', 'Unknown error')}")

            # 等待5秒后重试
            await asyncio.sleep(5)

    async def _download_result(self, result_url: str) -> str:
        """下载处理结果"""
        print(f"[MinerU] 下载结果: {result_url}")

        response = self.session.get(result_url)

        if response.status_code != 200:
            raise Exception(f"下载结果失败: {response.text}")

        return response.text

    def _parse_markdown(self, markdown_content: str) -> Dict[str, Any]:
        """
        解析 Markdown 内容

        提取：
        - 章节结构
        - 公式
        - 表格
        """
        import re

        result = {
            'markdown': markdown_content,
            'structure': [],
            'formulas': [],
            'tables': [],
            'images': []
        }

        lines = markdown_content.split('\n')
        current_section = None

        for line in lines:
            # 提取章节标题
            if line.startswith('#'):
                level = len(line) - len(line.lstrip('#'))
                title = line.lstrip('#').strip()
                current_section = {
                    'level': level,
                    'title': title
                }
                result['structure'].append(current_section)

            # 提取LaTeX公式
            # 行内公式: $...$
            inline_formulas = re.findall(r'\$([^$]+)\$', line)
            for formula in inline_formulas:
                result['formulas'].append({
                    'latex': formula,
                    'type': 'inline',
                    'context': line
                })

            # 块级公式: $$...$$
            block_formulas = re.findall(r'\$\$([^$]+)\$\$', line, re.DOTALL)
            for formula in block_formulas:
                result['formulas'].append({
                    'latex': formula.strip(),
                    'type': 'block',
```

## 性能优化

### 1. 缓存机制

```python
# utils/cache.py

import hashlib
import pickle
from typing import Any, Optional
import os

class DocumentCache:
    """文档处理缓存"""

    def __init__(self, cache_dir: str = './cache'):
        self.cache_dir = cache_dir
        os.makedirs(cache_dir, exist_ok=True)

    def get(self, file_path: str) -> Optional[Dict[str, Any]]:
        """获取缓存"""
        cache_key = self._get_cache_key(file_path)
        cache_file = os.path.join(self.cache_dir, f"{cache_key}.pkl")

        if os.path.exists(cache_file):
            print(f"[Cache] 命中缓存: {cache_key}")
            with open(cache_file, 'rb') as f:
                return pickle.load(f)

        return None

    def set(self, file_path: str, result: Dict[str, Any]):
        """设置缓存"""
        cache_key = self._get_cache_key(file_path)
        cache_file = os.path.join(self.cache_dir, f"{cache_key}.pkl")

        with open(cache_file, 'wb') as f:
            pickle.dump(result, f)

        print(f"[Cache] 已缓存: {cache_key}")

    def _get_cache_key(self, file_path: str) -> str:
        """生成缓存键（使用MD5）"""
        hash_md5 = hashlib.md5()
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(4096), b""):
                hash_md5.update(chunk)
        return hash_md5.hexdigest()

# 使用
class DocumentPreprocessor:
    def __init__(self):
        self.cache = DocumentCache()

    async def process(self, file_path: str, user_id: str) -> Dict[str, Any]:
        # 检查缓存
        cached_result = self.cache.get(file_path)
        if cached_result:
            return cached_result

        # 处理文档
        result = await self._process_without_cache(file_path, user_id)

        # 设置缓存
        self.cache.set(file_path, result)

        return result
```

```python
# 对于批量上传，使用异步并发

import asyncio

class BatchDocumentProcessor:
    """批量文档处理器"""

    def __init__(self):
        self.preprocessor = DocumentPreprocessor()
        self.max_concurrent = 5  # 最多同时处理5个文档

    async def process_batch(self, file_paths: List[str], user_id: str) -> List[Dict[str, Any]]:
        """批量处理文档"""
        semaphore = asyncio.Semaphore(self.max_concurrent)

        async def process_with_limit(file_path):
            async with semaphore:
                return await self.preprocessor.process(file_path, user_id)

        tasks = [process_with_limit(fp) for fp in file_paths]
        results = await asyncio.gather(*tasks)

        return results
```

```python
# 对于超大PDF（>50MB），分页处理

class LargeFileProcessor:
    """大文件处理器"""

    async def process_large_pdf(self, file_path: str, user_id: str) -> Dict[str, Any]:
        """处理大文件"""
        import fitz

        doc = fitz.open(file_path)
        total_pages = len(doc)

        # 按10页分片
        chunk_size = 10
        chunks = []

        for start in range(0, total_pages, chunk_size):
            end = min(start + chunk_size, total_pages)
            chunks.append((start, end))

        # 并发处理各片
        results = await self._process_chunks(file_path, chunks)

        # 合并结果
        merged_result = self._merge_results(results)

        return merged_result
```
