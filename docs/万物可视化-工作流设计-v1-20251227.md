## 五层处理架构

### 第一层：输入预处理层（InputPreprocessor）

**职责**: 接收多种输入类型，进行标准化处理

**输入类型支持**:
- 📄 PDF 文档
- 🖼️ 图片文件（JPG, PNG, 等）
- 📊 数据文件（CSV, JSON, Excel）
- 📝 纯文本
- 🔗 URL 链接

**处理流程**:

```python
class InputPreprocessor:
    def process(self, input_data):
        """
        标准化输入处理
        """
        # 1. 文件类型检测
        file_type = self.detect_type(input_data)

        # 2. 格式验证
        if not self.validate_format(input_data, file_type):
            raise InvalidFormatError(f"Invalid {file_type} format")

        # 3. 内容提取
        raw_content = self.extract_content(input_data, file_type)

        # 4. 元数据提取
        metadata = self.extract_metadata(input_data, file_type)

        return {
            'type': file_type,
            'content': raw_content,
            'metadata': metadata
        }
```

**输出格式**:
```json
{
  "type": "pdf|image|data|text|url",
  "content": "原始内容或二进制数据",
  "metadata": {
    "filename": "example.pdf",
    "size": 1024000,
    "pages": 15,
    "created": "2025-12-27",
    "language": "zh-CN"
  }
}
```

**职责**: 深度解析文档内容，提取关键信息

#### Agent 1: 文档解析器（DocumentParser）

**功能**: 逐页解析文档，提取文本和结构

```python
class DocumentParser:
    def parse_document(self, processed_input):
        """
        逐页解析文档
        """
        results = []

        # PDF 处理
        if processed_input['type'] == 'pdf':
            results = self.parse_pdf(processed_input['content'])

        # 图片处理
        elif processed_input['type'] == 'image':
            results = self.parse_image(processed_input['content'])

        # 数据文件处理
        elif processed_input['type'] == 'data':
            results = self.parse_data(processed_input['content'])

        return results

    def parse_pdf(self, pdf_content):
        """
        PDF 逐页解析
        """
        pages = []

        # 使用 PyMuPDF 或 pdfplumber
        with pdfplumber.open(pdf_content) as pdf:
            for page_num, page in enumerate(pdf.pages):
                page_data = {
                    'page_number': page_num + 1,
                    'text': page.extract_text(),
                    'images': page.extract_images(),
                    'tables': page.extract_tables(),
                    'layout': self.analyze_layout(page)
                }
                pages.append(page_data)

        return pages

    def parse_image(self, image_content):
        """
        图像内容识别
        """
        # 使用 OCR + 图像理解模型
        ocr_text = self.ocr_extraction(image_content)
        visual_understanding = self.vision_model_understanding(image_content)

        return {
            'text': ocr_text,
            'visual_description': visual_understanding,
            'elements': self.extract_visual_elements(image_content)
        }
```

#### Agent 2: 内容结构化器（ContentStructurer）

**功能**: 将解析的内容结构化，识别章节、图表、公式等

```python
class ContentStructurer:
    def structure_content(self, parsed_pages):
        """
        结构化内容
        """
        structured = {
            'title': '',
            'abstract': '',
            'sections': [],
            'figures': [],
            'tables': [],
            'formulas': [],
            'references': []
        }

        for page in parsed_pages:
            # 提取标题
            if not structured['title']:
                structured['title'] = self.extract_title(page)

            # 提取章节
            sections = self.extract_sections(page)
            structured['sections'].extend(sections)

            # 提取图片
            figures = self.extract_figures(page)
            structured['figures'].extend(figures)

            # 提取表格
            tables = self.extract_tables(page)
            structured['tables'].extend(tables)

            # 提取公式（LaTeX 格式）
            formulas = self.extract_formulas(page)
            structured['formulas'].extend(formulas)

        return structured

    def extract_sections(self, page):
        """
        提取章节结构
        """
        # 使用正则表达式或 NLP 模型识别章节标题
        patterns = [
            r'^第[一二三四五六七八九十]+章\s+.+$',
            r'^\d+\.\d+\s+.+$',
            r'^[A-Z][A-Z\s]+$'
        ]

        sections = []
        for line in page['text'].split('\n'):
            for pattern in patterns:
                if re.match(pattern, line):
                    sections.append({
                        'title': line.strip(),
                        'level': self.determine_level(line),
                        'page': page['page_number']
                    })
                    break

        return sections
```

#### Agent 4: 学科分类器（SubjectClassifier）

**功能**: 基于关键词判断学科

```python
class SubjectClassifier:
    def classify(self, structured_content, keywords):
        """
        学科分类
        """
        scores = {}

        # 计算每个学科的匹配分数
        for subject, domains in self.subject_keywords.items():
            subject_score = 0
            matched_domains = []

            for domain, terms in domains.items():
                domain_score = 0

                # 计算关键词匹配
                for keyword in keywords['subject_specific']:
                    if keyword in terms:
                        domain_score += 1

                # 检查标题和章节
                for section in structured_content['sections']:
                    if any(term in section['title'] for term in terms):
                        domain_score += 2

                if domain_score > 0:
                    subject_score += domain_score
                    matched_domains.append(domain)

            if subject_score > 0:
                scores[subject] = {
                    'score': subject_score,
                    'confidence': self.calculate_confidence(subject_score),
                    'matched_domains': matched_domains
                }

        # 排序并返回最佳匹配
        sorted_scores = sorted(
            scores.items(),
            key=lambda x: x[1]['score'],
            reverse=True
        )

        if not sorted_scores:
            return {
                'primary_subject': 'general',
                'confidence': 'low',
                'suggested_subjects': []
            }

        return {
            'primary_subject': sorted_scores[0][0],
            'confidence': sorted_scores[0][1]['confidence'],
            'matched_domains': sorted_scores[0][1]['matched_domains'],
            'suggested_subjects': [
                {
                    'subject': subj,
                    'score': data['score']
                }
                for subj, data in sorted_scores[1:4]
            ]
        }

    def calculate_confidence(self, score):
        """
        计算置信度
        """
        if score >= 10:
            return 'high'
        elif score >= 5:
            return 'medium'
        else:
            return 'low'
```

```python
class ConceptAnalyzer:
    def analyze_concepts(self, structured_content, subject_info):
        """
        分析核心概念
        """
        concepts = []

        # 从章节标题提取概念
        for section in structured_content['sections']:
            concept = {
                'name': section['title'],
                'type': self.determine_concept_type(section['title']),
                'level': section['level'],
                'importance': self.calculate_importance(section)
            }
            concepts.append(concept)

        # 从公式提取概念
        for formula in structured_content['formulas']:
            # 提取公式中的数学符号
            symbols = self.extract_mathematical_symbols(formula['latex'])
            concepts.append({
                'name': f"Formula: {formula['latex'][:50]}...",
                'type': 'formula',
                'symbols': symbols,
                'importance': 'high'
            })

        # 从图表说明提取概念
        for figure in structured_content['figures']:
            concepts.append({
                'name': figure['caption'],
                'type': 'visual',
                'importance': 'medium'
            })

        return concepts
```

#### Agent 6: 前置知识探索器（PrerequisiteExplorer）

**功能**: 递归探索概念的前置知识（核心创新）

```python
class PrerequisiteExplorer:
    def explore_prerequisites(self, concept, subject, depth=0, max_depth=3):
        """
        递归探索前置知识

        这是借鉴 Math2Manim 的核心创新！
        """
        if depth >= max_depth:
            return {
                'concept': concept,
                'prerequisites': [],
                'is_foundation': True
            }

        # 检查是否是基础概念
        if self.is_foundation_concept(concept, subject):
            return {
                'concept': concept,
                'prerequisites': [],
                'is_foundation': True
            }

        # 查找前置知识
        prereqs = self.find_prerequisites(concept, subject)

        # 递归探索每个前置知识
        prereq_trees = []
        for prereq in prereqs:
            prereq_tree = self.explore_prerequisites(
                prereq,
                subject,
                depth + 1,
                max_depth
            )
            prereq_trees.append(prereq_tree)

        return {
            'concept': concept,
            'prerequisites': prereq_trees,
            'is_foundation': False
        }

    def find_prerequisites(self, concept, subject):
        """
        使用 LLM 查找前置知识
        """
        prompt = f"""
        学科: {subject}
        概念: {concept}

        问题: 要理解这个概念，必须先掌握哪些前置知识？
        请列出 3-5 个最关键的前置概念。

        只返回概念列表，用逗号分隔。
        """

        response = self.llm_client.generate(prompt)

        # 解析并返回前置概念列表
        return [p.strip() for p in response.split(',')]
```


#### Agent 8: 可视化生成器（VisualizationRenderer）

**功能**: 生成可视化代码并执行

```python
class VisualizationRenderer:
    def generate(self, concept_tree, template, subject):
        """
        生成可视化
        """
        # 1. 构建详细提示词（借鉴 Math2Manim 的 LaTeX 富提示）
        detailed_prompt = self.build_detailed_prompt(
            concept_tree,
            template
        )

        # 2. 选择合适的渲染引擎
        if subject in ['mathematics', 'physics', 'chemistry']:
            render_engine = 'manim'  # 数学动画
        elif subject in ['computer_science', 'data_science']:
            render_engine = 'd3.js'  # 交互式可视化
        elif subject in ['biology', 'medicine']:
            render_engine = 'three.js'  # 3D 分子模型
        else:
            render_engine = 'generic'  # 通用图表

        # 3. 生成代码
        code = self.generate_code(detailed_prompt, render_engine)

        # 4. 执行渲染
        output = self.render(code, render_engine)

        return {
            'code': code,
            'output': output,
            'engine': render_engine
        }

    def build_detailed_prompt(self, concept_tree, template):
        """
        构建详细提示词（LaTeX 富提示策略）
        """
        prompt = f"""
# 可视化生成请求

## 目标概念
{concept_tree['concept']}

## 学科领域
{template['subject']}

## 模板规范
- 模板名称: {template['template']}
- 视觉元素: {', '.join(template['visual_elements'])}
- 动画效果: {', '.join(template['animations'])}

## 前置知识路径
"""

        # 添加前置知识路径
        def add_prereqs(tree, level=0):
            indent = "  " * level
            prompt += f"{indent}- {tree['concept']}\n"
            for prereq in tree['prerequisites']:
                add_prereqs(prereq, level + 1)

        add_prereqs(concept_tree)

        # 添加详细的视觉规范
        prompt += f"""

## 详细视觉规范

### 颜色方案
- 主色调: {template.get('color_scheme', {}).get('primary', '#3498db')}
- 辅助色: {template.get('color_scheme', {}).get('secondary', '#e74c3c')}
- 背景色: {template.get('color_scheme', {}).get('background', '#ffffff')}

### 动画时序
"""
        # 添加动画时间规划
        total_duration = template.get('duration', 30)
        num_concepts = self.count_concepts(concept_tree)
        time_per_concept = total_duration / num_concepts

        prompt += f"- 总时长: {total_duration} 秒\n"
        prompt += f"- 每个概念: 约 {time_per_concept:.1f} 秒\n"

        prompt += """

### 代码生成要求
1. 使用标准的渲染库API
2. 添加详细的注释
3. 确保动画流畅过渡
4. 使用 LaTeX 格式的数学公式（如需要）
5. 代码必须可以直接运行，无需额外资源

请根据以上规范生成完整的可视化代码。
"""

        return prompt

    def generate_code(self, prompt, engine):
        """
        使用 LLM 生成代码
        """
        # 调用 LLM API
        response = self.llm_client.generate(
            prompt=prompt,
            model="claude-sonnet-4-5",  # 或其他模型
            max_tokens=4000
        )

        # 提取代码块
        code = self.extract_code_block(response)

        return code
```

```python
# backend/main.py
from fastapi import FastAPI, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware

app = FastAPI(title="万物可视化 AI")

# CORS 配置
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# 导入 Agent
from agents import (
    InputPreprocessor,
    DocumentParser,
    ContentStructurer,
    KeywordExtractor,
    SubjectClassifier,
    ConceptAnalyzer,
    PrerequisiteExplorer,
    TemplateMatcher,
    VisualizationRenderer
)

@app.post("/api/analyze")
async def analyze_document(file: UploadFile = File(...)):
    """
    主端点：处理文档并生成可视化
    """
    # 第一层：输入预处理
    preprocessor = InputPreprocessor()
    processed = await preprocessor.process(file)

    # 第二层：内容分析
    parser = DocumentParser()
    parsed = parser.parse_document(processed)

    structurer = ContentStructurer()
    structured = structurer.structure_content(parsed)

    # 第三层：学科识别
    keyword_extractor = KeywordExtractor()
    keywords = keyword_extractor.extract_keywords(structured)

    classifier = SubjectClassifier()
    subject_info = classifier.classify(structured, keywords)

    # 第四层：知识构建
    concept_analyzer = ConceptAnalyzer()
    concepts = concept_analyzer.analyze_concepts(structured, subject_info)

    prerequisite_explorer = PrerequisiteExplorer()
    knowledge_tree = prerequisite_explorer.explore_prerequisites(
        concepts[0],  # 主概念
        subject_info['primary_subject']
    )

    # 第五层：可视化生成
    template_matcher = TemplateMatcher()
    template = template_matcher.match_template(
        concepts[0],
        subject_info['primary_subject'],
        subject_info['matched_domains'][0]
    )

    renderer = VisualizationRenderer()
    visualization = renderer.generate(
        knowledge_tree,
        template,
        subject_info['primary_subject']
    )

    return {
        'subject_info': subject_info,
        'concepts': concepts,
        'knowledge_tree': knowledge_tree,
        'visualization': visualization
    }
```

```javascript
// frontend-v4/js/api.js

class VisualizationAPI {
    constructor(baseURL = 'http://localhost:8000') {
        this.baseURL = baseURL;
    }

    async analyzeDocument(file) {
        const formData = new FormData();
        formData.append('file', file);

        try {
            // 显示加载状态
            this.showLoading('正在分析文档...');

            // 发送请求
            const response = await fetch(`${this.baseURL}/api/analyze`, {
                method: 'POST',
                body: formData
            });

            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }

            const result = await response.json();

            // 显示结果
            this.displayResults(result);

            return result;
        } catch (error) {
            console.error('Error:', error);
            this.showError('分析失败: ' + error.message);
        }
    }

    displayResults(result) {
        // 1. 显示学科识别结果
        this.showSubjectInfo(result.subject_info);

        // 2. 显示概念列表
        this.showConcepts(result.concepts);

        // 3. 显示知识树
        this.showKnowledgeTree(result.knowledge_tree);

        // 4. 显示可视化结果
        this.showVisualization(result.visualization);
    }

    showSubjectInfo(subjectInfo) {
        const container = document.getElementById('subjectInfo');

        container.innerHTML = `
            <h3>识别的学科</h3>
            <div class="subject-badge primary">
                ${this.translateSubject(subjectInfo.primary_subject)}
                <span class="confidence ${subjectInfo.confidence}">
                    置信度: ${subjectInfo.confidence}
                </span>
            </div>
            <div class="matched-domains">
                ${subjectInfo.matched_domains.map(d =>
                    `<span class="domain-tag">${d}</span>`
                ).join('')}
            </div>
        `;
    }

    showVisualization(visualization) {
        const container = document.getElementById('visualizationPreview');

        // 根据渲染引擎类型显示结果
        if (visualization.engine === 'manim') {
            // 显示视频
            container.innerHTML = `
                <video controls>
                    <source src="${visualization.output.video_url}" type="video/mp4">
                </video>
            `;
        } else if (visualization.engine === 'd3.js') {
            // 渲染 D3.js 可视化
            this.renderD3Visualization(visualization.code, container);
        } else if (visualization.engine === 'three.js') {
            // 渲染 Three.js 3D 场景
            this.renderThreeVisualization(visualization.code, container);
        }

        // 显示代码
        const codeContainer = document.getElementById('codePreview');
        codeContainer.textContent = visualization.code;
    }

            <html>

                <script src="https://d3js.org/d3.v7.min.js"></script>
                <style>
                    body { margin: 0; }
                    svg { width: 100%; height: 100%; }
                </style>
            </head>
            <body>
                <div id="viz"></div>
                <script>
                    ${code}
                </script>
            </body>
            </html>
        `;

        container.appendChild(iframe);
        iframe.srcdoc = html;
    }
```
