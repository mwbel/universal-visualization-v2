## 三种处理策略

### 策略1: 小文档 - 大模型直接处理（推荐）⭐

**适用场景：**
- 文档大小 < 20页
- PDF、图片、文本都可以
- 需要快速响应

**处理流程：**
```python
class DirectLLMAnalyzer:
    """直接使用大模型分析"""

    def analyze(self, file):
        # 1. 转换为多模态输入
        if file.type == 'pdf':
            # 转换PDF为图片（每页）
            images = self.pdf_to_images(file)
            content = {
                'type': 'multimodal',
                'images': images
            }
        elif file.type == 'image':
            content = {
                'type': 'multimodal',
                'images': [file]
            }

        # 2. 一次性调用 Gemini 1.5 Pro
        prompt = self._build_comprehensive_prompt()

        result = self.llm_client.generate_multimodal(
            prompt=prompt,
            images=content['images'],
            model='gemini-1.5-pro'
        )

        # 3. 解析结构化结果
        return self._parse_structured_result(result)

    def _build_comprehensive_prompt(self):
        return """
请分析这份文档，提供以下信息（JSON格式）：

1. **基本信息**
   - 文档标题
   - 作者/来源
   - 摘要（100字以内）

2. **结构信息**
   - 章节列表（带层级）
   - 页码范围

3. **内容分析**
   - 核心概念（按重要性排序，前10个）
   - 关键公式（如有，LaTeX格式）
   - 重要图表（带描述）

4. **学科识别**
   - 主要学科（mathematics/physics/chemistry等）
   - 相关领域（calculus/quantum/organic等）
   - 置信度（high/medium/low）
   - 匹配的关键词

5. **前置知识树**（目标概念的前置知识）
   - 从目标概念递归到基础
   - 最多3层深度

请以JSON格式返回，确保可解析。
"""
```

**优点：**
- 最快最简单
- 全局理解，不会丢失上下文
- 一次调用完成所有分析

**示例代码：**

```python
# agents/llm_analyzer.py

import base64
import json
from google import genai
from typing import Dict, List, Any

class MultiModalAnalyzer:
    """多模态大模型分析器"""

    def __init__(self):
        self.client = genai.Client(api_key=os.getenv("GOOGLE_API_KEY"))

    def analyze_document(
        self,
        file_path: str,
        file_type: str,
        max_tree_depth: int = 3
    ) -> Dict[str, Any]:
        """
        使用 Gemini 1.5 Pro 一步到位分析文档
        """
        # 1. 准备输入
        if file_type == 'pdf':
            # PDF转图片
            images = self._convert_pdf_to_images(file_path)
        elif file_type in ['jpg', 'png', 'jpeg']:
            images = [self._load_image(file_path)]
        else:
            # 文本文件
            with open(file_path, 'r', encoding='utf-8') as f:
                text_content = f.read()
            images = []

        # 2. 构建提示词
        prompt = self._build_analysis_prompt(max_tree_depth)

        # 3. 调用 Gemini
        response = self._call_gemini(prompt, images, text_content if not images else None)

        # 4. 解析结果
        result = self._parse_response(response)

        return result

    def _convert_pdf_to_images(self, pdf_path: str) -> List[str]:
        """将PDF转换为图片（base64）"""
        import fitz  # PyMuPDF

        doc = fitz.open(pdf_path)
        images = []

        for page_num in range(min(len(doc), 50)):  # 限制最多50页
            page = doc[page_num]
            pix = page.get_pixmap(matrix=fitz.Matrix(2, 2))  # 2x缩放
            img_bytes = pix.tobytes("png")
            img_base64 = base64.b64encode(img_bytes).decode('utf-8')
            images.append(img_base64)

        doc.close()
        return images

    def _build_analysis_prompt(self, max_depth: int) -> str:
        return f"""
请详细分析这份文档，并以JSON格式返回以下信息：

{{
  "basic_info": {{
    "title": "文档标题",
    "authors": ["作者列表"],
    "abstract": "摘要（100字以内）"
  }},

  "structure": {{
    "sections": [
      {{"level": 1, "title": "第一章", "page": 1}},
      {{"level": 2, "title": "1.1 引言", "page": 2}}
    ],
    "total_pages": "总页数"
  }},

  "content_analysis": {{
    "main_concepts": [
      {{"name": "概念名", "importance": "high/medium/low", "description": "简短描述"}}
    ],
    "formulas": [
      {{"latex": "公式LaTeX", "description": "公式含义", "page": 页码}}
    ],
    "figures": [
      {{"caption": "图表标题", "description": "内容描述", "page": 页码}}
    ],
    "tables": [
      {{"caption": "表格标题", "rows": 行数, "page": 页码}}
    ]
  }},

  "subject_classification": {{
    "primary_subject": "mathematics/physics/chemistry/biology/computer_science/economics/other",
    "domains": ["calculus", "geometry", ...],
    "confidence": "high/medium/low",
    "matched_keywords": ["关键词1", "关键词2"]
  }},

  "knowledge_tree": {{
    "target_concept": "文档的核心目标概念",
    "tree": {{
      "concept": "目标概念",
      "prerequisites": [
        {{
          "concept": "前置概念1",
          "prerequisites": [
            {{"concept": "基础概念", "prerequisites": [], "is_foundation": true}}
          ]
        }}
      ]
    }}
  }}
}}

**重要要求：**
1. 确保返回的是有效的JSON格式
2. knowledge_tree 的深度不超过 {max_depth} 层
3. concepts 最多返回10个最重要的
4. subject_classification 必须从预定义的学科中选择
5. 如果某个字段无法提取，使用空数组或空字符串

请开始分析：
"""

    def _call_gemini(
        self,
        prompt: str,
        images: List[str] = None,
        text: str = None
    ) -> str:
        """调用 Gemini API"""

        # 构建内容
        contents = [{"text": prompt}]

        # 添加图片
        if images:
            for img_base64 in images:
                contents.append({
                    "inline_data": {
                        "mime_type": "image/png",
                        "data": img_base64
                    }
                })

        # 添加文本
        if text:
            contents[0]["text"] += f"\n\n文档内容：\n{text}"

        try:
            response = self.client.models.generate_content(
                model='gemini-1.5-pro',
                contents=contents
            )
            return response.text

        except Exception as e:
            print(f"Gemini API 调用失败: {e}")
            raise

    def _parse_response(self, response: str) -> Dict[str, Any]:
        """解析 Gemini 响应"""
        # 提取JSON
        import re

        # 尝试直接解析
        try:
            return json.loads(response)
        except:
            pass

        # 尝试提取JSON块
        json_match = re.search(r'```json\n(.*?)\n```', response, re.DOTALL)
        if json_match:
            try:
                return json.loads(json_match.group(1))
            except:
                pass

        # 尝试提取 { ... }
        json_match = re.search(r'\{.*\}', response, re.DOTALL)
        if json_match:
            try:
                return json.loads(json_match.group(0))
            except:
                pass

        # 如果都失败，返回原始响应
        return {
            "raw_response": response,
            "parse_error": "无法解析为JSON"
        }
```

### 策略2: 大文档 - 分段+大模型

**适用场景：**
- 文档大小 > 20页
- 需要详细分析
- 控制成本

**处理流程：**
```python
class ChunkedLLMAnalyzer:
    """分段分析+汇总"""

    def analyze(self, file):
        # 1. 按章节分段
        chunks = self.split_by_chapters(file)

        # 2. 并行分析每段
        chunk_results = self._analyze_chunks_parallel(chunks)

        # 3. 汇总全局视图
        summary = self._summarize_results(chunk_results)

        return summary

    def split_by_chapters(self, file):
        """智能分段：按章节、不超过token限制"""
        # 使用轻量级工具识别章节结构
        structure = self._detect_structure(file)

        chunks = []
        for chapter in structure.chapters:
            chunk = {
                'content': file.extract(chapter.start, chapter.end),
                'metadata': {
                    'chapter': chapter.title,
                    'page_range': (chapter.start, chapter.end)
                }
            }
            chunks.append(chunk)

        return chunks

    def _analyze_chunks_parallel(self, chunks):
        """并行分析各段"""
        import asyncio

        async def analyze_chunk(chunk):
            prompt = f"""
分析这一章节，提供：
1. 核心概念（5个）
2. 关键公式
3. 本章涉及的学科领域

章节标题：{chunk['metadata']['chapter']}
内容：{chunk['content'][:10000]}  # 限制长度
"""
            result = await self.llm_client.generate_async(prompt)
            return {
                'chunk': chunk['metadata'],
                'analysis': result
            }

        # 并行执行
        results = asyncio.run(analyze_chunk(chunks))
        return results

    def _summarize_results(self, chunk_results):
        """汇总成全局视图"""
        # 提取所有概念
        all_concepts = []
        for result in chunk_results:
            all_concepts.extend(result['concepts'])

        # 去重和排序
        unique_concepts = self._deduplicate_concepts(all_concepts)

        # 识别主要学科
        subject_votes = {}
        for result in chunk_results:
            for subj in result['subjects']:
                subject_votes[subj] = subject_votes.get(subj, 0) + 1

        primary_subject = max(subject_votes, key=subject_votes.get)

        return {
            'concepts': unique_concepts,
            'primary_subject': primary_subject,
            'subject_confidence': subject_votes[primary_subject] / len(chunk_results)
        }
```

### 策略3: 成本敏感场景 - 传统工具+大模型验证

**适用场景：**
- 批量处理大量文档
- 成本敏感
- 对速度要求不高

**处理流程：**
```python
class HybridAnalyzer:
    """传统工具为主，大模型验证"""

    def analyze(self, file):
        # 1. 使用传统工具提取
        extracted = self._extract_with_tools(file)

        # 2. 质量检查
        quality_score = self._assess_quality(extracted)

        # 3. 如果质量不达标，调用大模型
        if quality_score < 0.7:
            extracted = self._llm_refine(file, extracted)

        return extracted

    def _extract_with_tools(self, file):
        """使用传统工具提取"""
        # PDF → pdfplumber
        # 图片 → pytesseract
        # 表格 → camelot
        # 公式 → LaTeX OCR

        result = {
            'text': self._extract_text(file),
            'tables': self._extract_tables(file),
            'formulas': self._extract_formulas(file),
            'structure': self._detect_structure(file)
        }

        return result

    def _assess_quality(self, extracted):
        """评估提取质量"""
        score = 1.0

        # 检查文本提取率
        if len(extracted['text']) < 100:
            score -= 0.3

        # 检查是否有章节结构
        if not extracted['structure'].get('sections'):
            score -= 0.2

        # 检查OCR准确率（通过语言模型检测）
        if self._has_ocr_errors(extracted['text']):
            score -= 0.3

        return max(0, score)

    def _llm_refine(self, file, initial_extract):
        """使用大模型优化"""
        prompt = f"""
我们已经使用工具提取了以下信息：

{json.dumps(initial_extract, ensure_ascii=False, indent=2)}

请：
1. 修正OCR错误
2. 补充遗漏的章节
3. 提取更准确的公式
4. 识别学科和概念

返回优化后的JSON。
"""

        response = self.llm_client.generate(prompt)
        return json.loads(response)
```

## 决策树：自动选择策略

```python
class SmartContentAnalyzer:
    """智能选择最佳分析策略"""

    def analyze(self, file):
        # 1. 评估文档特征
        features = self._assess_document(file)

        # 2. 选择策略
        if features.size < 20 and features.has_images:
            # 小文档 + 图片 → 大模型直接处理
            return self.direct_llm_analyzer.analyze(file)

        elif features.size > 20:
            # 大文档 → 分段处理
            if features.budget == 'high':
                return self.chunked_llm_analyzer.analyze(file)
            else:
                return self.hybrid_analyzer.analyze(file)

        else:
            # 普通文档 → 混合方案
            return self.hybrid_analyzer.analyze(file)

    def _assess_document(self, file):
        """评估文档特征"""
        return DocumentFeatures(
            size=self._count_pages(file),
            has_images=self._has_images(file),
            has_tables=self._has_tables(file),
            complexity=self._estimate_complexity(file),
            budget=self._get_user_budget()  # 'low', 'medium', 'high'
        )
```

```python
# agents/content_analyzer.py

from typing import Dict, Any
from enum import Enum

class AnalysisStrategy(Enum):
    DIRECT_LLM = "direct_llm"
    CHUNKED_LLM = "chunked_llm"
    HYBRID = "hybrid"

class ContentAnalyzer:
    """统一的内容分析接口"""

    def __init__(self, strategy: AnalysisStrategy = AnalysisStrategy.DIRECT_LLM):
        self.strategy = strategy
        self.analyzers = {
            AnalysisStrategy.DIRECT_LLM: DirectLLMAnalyzer(),
            AnalysisStrategy.CHUNKED_LLM: ChunkedLLMAnalyzer(),
            AnalysisStrategy.HYBRID: HybridAnalyzer()
        }

    def analyze(self, file) -> Dict[str, Any]:
        """分析文档（根据策略自动选择）"""
        # 如果策略是自动，先评估
        if self.strategy == "auto":
            strategy = self._select_strategy(file)
        else:
            strategy = self.strategy

        # 使用对应的分析器
        analyzer = self.analyzers[strategy]
        return analyzer.analyze(file)

    def _select_strategy(self, file) -> AnalysisStrategy:
        """智能选择策略"""
        features = self._assess_file(file)

        if features.page_count < 20:
            return AnalysisStrategy.DIRECT_LLM
        elif features.page_count < 100:
            return AnalysisStrategy.CHUNKED_LLM
        else:
            return AnalysisStrategy.HYBRID

    def _assess_file(self, file):
        """评估文件特征"""
        return FileFeatures(
            page_count=self._count_pages(file),
            has_images=self._check_images(file),
            user_budget=self._get_user_budget()
        )
```

**总结：大模型一步到位更适合你的场景！**

**为什么？**
1. Gemini 1.5 Pro、Claude 3.5 Sonnet等模型已经非常强大
2. 多步骤分解会导致误差累积
3. 维护成本高

只有在以下情况才考虑多步骤：
- 超大规模批量处理（>1000篇/天）
- 极度成本敏感
- 需要实时流式处理

对于"万物可视化"这种场景，**直接用大模型**！
