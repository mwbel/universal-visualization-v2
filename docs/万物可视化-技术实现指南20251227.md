```python
@app.post("/api/v1/analyze")
async def analyze_document(
    file: UploadFile = File(..., description="上传的文档文件（PDF, 图片, 文本等）"),
    max_depth: int = 3,
    language: str = "zh-CN"
):
    """
    分析文档并生成可视化

    ## 文件类型支持
    - PDF 文档 (.pdf)
    - 图片文件 (.jpg, .png, .gif)
    - 文本文件 (.txt)
    - 数据文件 (.csv, .json)

    ## 返回结果
    - 学科识别信息
    - 概念列表
    - 知识树
    - 可视化代码和结果
    """
    try:
        logger.info(f"开始分析文档: {file.filename}")

        # ========== 第一层：输入预处理 ==========
        logger.info("步骤 1/8: 输入预处理")
        processed_input = await preprocessor.process(file)
        logger.info(f"文件类型: {processed_input['type']}, 大小: {processed_input['metadata']['size']}")

        # ========== 第二层：文档解析 ==========
        logger.info("步骤 2/8: 文档解析")
        parsed_content = parser.parse_document(processed_input)
        logger.info(f"解析了 {len(parsed_content.get('pages', []))} 页")

        # ========== 第二层（续）：内容结构化 ==========
        logger.info("步骤 3/8: 内容结构化")
        structured_content = structurer.structure_content(parsed_content)
        logger.info(f"提取了 {len(structured_content['sections'])} 个章节")

        # ========== 第三层：关键词提取 ==========
        logger.info("步骤 4/8: 关键词提取")
        keywords = keyword_extractor.extract_keywords(structured_content)
        logger.info(f"提取了 {len(keywords['subject_specific'])} 个学科关键词")

        # ========== 第三层（续）：学科分类 ==========
        logger.info("步骤 5/8: 学科分类")
        subject_info = classifier.classify(structured_content, keywords)
        logger.info(f"识别的学科: {subject_info['primary_subject']}, 置信度: {subject_info['confidence']}")

        # ========== 第四层：概念分析 ==========
        logger.info("步骤 6/8: 概念分析")
        concepts = concept_analyzer.analyze_concepts(
            structured_content,
            subject_info
        )
        logger.info(f"识别了 {len(concepts)} 个核心概念")

        # ========== 第四层（续）：前置知识探索 ==========
        logger.info("步骤 7/8: 前置知识探索")
        if concepts:
            main_concept = concepts[0]['name']
            knowledge_tree = prerequisite_explorer.explore_prerequisites(
                main_concept,
                subject_info['primary_subject'],
                max_depth=max_depth
            )
            logger.info(f"构建了知识树，深度: {max_depth}")
        else:
            knowledge_tree = None
            logger.warning("未找到核心概念，跳过知识树构建")

        # ========== 第五层：模板匹配 ==========
        logger.info("步骤 8/8: 可视化生成")
        if knowledge_tree and subject_info['matched_domains']:
            template = template_matcher.match_template(
                concepts[0] if concepts else None,
                subject_info['primary_subject'],
                subject_info['matched_domains'][0]
            )

            # 生成可视化
            visualization = renderer.generate(
                knowledge_tree,
                template,
                subject_info['primary_subject']
            )
            logger.info("可视化生成完成")
        else:
            template = None
            visualization = None
            logger.warning("无法生成可视化，缺少必要信息")

        # ========== 返回结果 ==========
        result = {
            "success": True,
            "file_info": {
                "filename": file.filename,
                "type": processed_input['type'],
                "size": processed_input['metadata']['size']
            },
            "subject_info": subject_info,
            "content_summary": {
                "total_pages": len(parsed_content.get('pages', [])),
                "total_sections": len(structured_content['sections']),
                "total_figures": len(structured_content['figures']),
                "total_tables": len(structured_content['tables']),
                "total_formulas": len(structured_content['formulas'])
            },
            "keywords": keywords,
            "concepts": concepts[:10],  # 限制返回前 10 个
            "knowledge_tree": knowledge_tree,
            "template": template,
            "visualization": visualization
        }

        logger.info(f"文档分析完成: {file.filename}")
        return result

    except Exception as e:
        logger.error(f"分析失败: {str(e)}", exc_info=True)
        raise HTTPException(status_code=500, detail=f"分析失败: {str(e)}")
```

```python
# agents/input_preprocessor.py

import aiofiles
from typing import Dict, Any
import magic  # python-magic 库用于文件类型检测
import os

class InputPreprocessor:
    """输入预处理 Agent"""

    async def process(self, file) -> Dict[str, Any]:
        """
        处理上传的文件

        Args:
            file: FastAPI UploadFile 对象

        Returns:
            处理后的输入数据字典
        """
        # 读取文件内容
        content = await file.read()

        # 检测文件类型
        file_type = self.detect_type(content, file.filename)

        # 验证文件
        self.validate_file(content, file_type)

        # 提取元数据
        metadata = self.extract_metadata(file, content)

        return {
            'type': file_type,
            'content': content,
            'metadata': metadata
        }

    def detect_type(self, content: bytes, filename: str) -> str:
        """检测文件类型"""
        # 使用 magic 库检测 MIME 类型
        mime = magic.from_buffer(content, mime=True)

        # 根据 MIME 类型映射到内部类型
        type_mapping = {
            'application/pdf': 'pdf',
            'image/jpeg': 'image',
            'image/png': 'image',
            'image/gif': 'image',
            'text/plain': 'text',
            'application/json': 'data',
            'text/csv': 'data',
            'application/vnd.ms-excel': 'data'
        }

        file_type = type_mapping.get(mime, 'unknown')

        if file_type == 'unknown':
            # 根据文件扩展名猜测
            ext = os.path.splitext(filename)[1].lower()
            ext_mapping = {
                '.pdf': 'pdf',
                '.jpg': 'image',
                '.jpeg': 'image',
                '.png': 'image',
                '.txt': 'text',
                '.json': 'data',
                '.csv': 'data'
            }
            file_type = ext_mapping.get(ext, 'unknown')

        return file_type

    def validate_file(self, content: bytes, file_type: str):
        """验证文件"""
        from config import settings

        # 检查文件大小
        if len(content) > settings.MAX_UPLOAD_SIZE:
            raise ValueError(f"文件过大，最大支持 {settings.MAX_UPLOAD_SIZE / 1024 / 1024} MB")

        # 检查文件类型
        if file_type == 'unknown':
            raise ValueError("不支持的文件类型")

    def extract_metadata(self, file, content: bytes) -> Dict[str, Any]:
        """提取元数据"""
        return {
            'filename': file.filename,
            'size': len(content),
            'content_type': file.content_type,
            'language': 'zh-CN'  # 默认中文，后续可以自动检测
        }
```

#### 3.2 文档解析 Agent

```python
# agents/document_parser.py

import pdfplumber
import pytesseract
from PIL import Image
import io
from typing import Dict, List, Any
import fitz  # PyMuPDF

class DocumentParser:
    """文档解析 Agent"""

    def __init__(self):
        self.ocr_lang = 'chi_sim+eng'  # 中文+英文

    def parse_document(self, processed_input: Dict[str, Any]) -> Dict[str, Any]:
        """解析文档"""
        file_type = processed_input['type']
        content = processed_input['content']

        if file_type == 'pdf':
            return self.parse_pdf(content)
        elif file_type == 'image':
            return self.parse_image(content)
        elif file_type == 'text':
            return self.parse_text(content)
        elif file_type == 'data':
            return self.parse_data(content)
        else:
            raise ValueError(f"不支持的文件类型: {file_type}")

    def parse_pdf(self, content: bytes) -> Dict[str, Any]:
        """解析 PDF 文档"""
        pdf_file = io.BytesIO(content)
        pages_data = []

        with pdfplumber.open(pdf_file) as pdf:
            for page_num, page in enumerate(pdf.pages):
                # 提取文本
                text = page.extract_text() or ""

                # 提取图片
                images = self.extract_pdf_images(content, page_num)

                # 提取表格
                tables = page.extract_tables() or []

                # 分析布局
                layout = self.analyze_layout(page)

                page_data = {
                    'page_number': page_num + 1,
                    'text': text,
                    'images': images,
                    'tables': tables,
                    'layout': layout
                }
                pages_data.append(page_data)

        return {
            'type': 'pdf',
            'total_pages': len(pages_data),
            'pages': pages_data
        }

    def extract_pdf_images(self, pdf_content: bytes, page_num: int) -> List[Dict]:
        """从 PDF 中提取图片"""
        doc = fitz.open(stream=pdf_content, filetype="pdf")
        page = doc[page_num]
        image_list = page.get_images(full=True)

        images = []
        for img_index, img in enumerate(image_list):
            xref = img[0]
            base_image = doc.extract_image(xref)
            image_bytes = base_image["image"]
            image_ext = base_image["ext"]

            # 保存图片到临时目录
            import tempfile
            import os

            temp_dir = tempfile.gettempdir()
            image_path = os.path.join(temp_dir, f"page_{page_num}_img_{img_index}.{image_ext}")

            with open(image_path, "wb") as f:
                f.write(image_bytes)

            images.append({
                'index': img_index,
                'path': image_path,
                'format': image_ext
            })

        doc.close()
        return images

    def parse_image(self, content: bytes) -> Dict[str, Any]:
        """解析图片"""
        image = Image.open(io.BytesIO(content))

        # OCR 提取文本
        text = pytesseract.image_to_string(image, lang=self.ocr_lang)

        # 使用视觉模型理解图片内容
        visual_description = self.vision_understanding(content)

        return {
            'type': 'image',
            'text': text,
            'visual_description': visual_description,
            'format': image.format,
            'size': image.size
        }

    def vision_understanding(self, image_content: bytes) -> str:
        """使用视觉模型理解图片"""
        # 这里可以调用 GPT-4V 或 Gemini Vision
        # 示例使用 OpenAI Vision API
        import base64
        from openai import OpenAI
        from config import settings

        if not settings.OPENAI_API_KEY:
            return ""

        client = OpenAI(api_key=settings.OPENAI_API_KEY)

        # 编码图片
        base64_image = base64.b64encode(image_content).decode('utf-8')

        try:
            response = client.chat.completions.create(
                model="gpt-4-vision-preview",
                messages=[
                    {
                        "role": "user",
                        "content": [
                            {
                                "type": "text",
                                "text": "请详细描述这张图片的内容，包括文字、图表、公式等。"
                            },
                            {
                                "type": "image_url",
                                "image_url": {
                                    "url": f"data:image/jpeg;base64,{base64_image}"
                                }
                            }
                        ]
                    }
                ],
                max_tokens=500
            )

            return response.choices[0].message.content

        except Exception as e:
            print(f"Vision API 调用失败: {e}")
            return ""

    def parse_text(self, content: bytes) -> Dict[str, Any]:
        """解析文本"""
        text = content.decode('utf-8')

        # 按段落分割
        paragraphs = [p.strip() for p in text.split('\n\n') if p.strip()]

        return {
            'type': 'text',
            'text': text,
            'paragraphs': paragraphs,
            'total_paragraphs': len(paragraphs)
        }

    def parse_data(self, content: bytes) -> Dict[str, Any]:
        """解析数据文件"""
        import json
        import csv
        import pandas as pd
        import io

        # 尝试 JSON
        try:
            data = json.loads(content.decode('utf-8'))
            return {
                'type': 'json',
                'data': data
            }
        except:
            pass

        # 尝试 CSV
        try:
            df = pd.read_csv(io.BytesIO(content))
            return {
                'type': 'csv',
                'data': df.to_dict('records'),
                'columns': df.columns.tolist(),
                'shape': df.shape
            }
        except:
            pass

        raise ValueError("无法解析数据文件")

    def analyze_layout(self, page) -> Dict[str, Any]:
        """分析页面布局"""
        # 简单的布局分析
        text = page.extract_text() or ""

        # 识别标题
        lines = text.split('\n')
        titles = [line for line in lines if len(line) < 50 and line.strip()]

        # 识别是否有图表
        has_images = len(page.images) > 0
        has_tables = page.extract_tables() is not None and len(page.extract_tables()) > 0

        return {
            'titles': titles,
            'has_images': has_images,
            'has_tables': has_tables,
            'line_count': len(lines)
        }
```

```python
    def classify(self, structured_content: Dict[str, Any],
                 keywords: Dict[str, List[str]]) -> Dict[str, Any]:
        """分类文档"""
        scores = {}

        # 计算每个学科的分数
        for subject, domains in self.subject_keywords.items():
            subject_score = 0
            matched_domains = []
            matched_keywords = []

            for domain, terms in domains.items():
                domain_score = 0
                domain_keywords = []

                # 检查关键词匹配
                for keyword in keywords['subject_specific']:
                    if any(term.lower() in keyword.lower() for term in terms):
                        domain_score += 1
                        domain_keywords.append(keyword)

                # 检查标题和章节
                for section in structured_content.get('sections', []):
                    if any(term.lower() in section['title'].lower() for term in terms):
                        domain_score += 2

                if domain_score > 0:
                    subject_score += domain_score
                    matched_domains.append(domain)
                    matched_keywords.extend(domain_keywords)

            if subject_score > 0:
                scores[subject] = {
                    'score': subject_score,
                    'confidence': self._calculate_confidence(subject_score),
                    'matched_domains': matched_domains,
                    'matched_keywords': list(set(matched_keywords))
                }

        # 排序
        sorted_scores = sorted(scores.items(), key=lambda x: x[1]['score'], reverse=True)

        if not sorted_scores:
            return {
                'primary_subject': 'general',
                'confidence': 'low',
                'matched_domains': [],
                'suggested_subjects': []
            }

        return {
            'primary_subject': sorted_scores[0][0],
            'confidence': sorted_scores[0][1]['confidence'],
            'matched_domains': sorted_scores[0][1]['matched_domains'],
            'matched_keywords': sorted_scores[0][1]['matched_keywords'],
            'suggested_subjects': [
                {
                    'subject': subj,
                    'score': data['score']
                }
                for subj, data in sorted_scores[1:4]
            ]
        }

    def _calculate_confidence(self, score: int) -> str:
        """计算置信度"""
        if score >= 10:
            return 'high'
        elif score >= 5:
            return 'medium'
        else:
            return 'low'
```

```python
class PrerequisiteExplorer:
    """前置知识探索 Agent（借鉴 Math2Manim 的核心创新）"""

    def __init__(self):

        self.foundation_concepts = self._load_foundation_concepts()

    def _load_foundation_concepts(self) -> Dict[str, List[str]]:
        """加载基础概念列表"""
        return {
            'mathematics': [
                '数', '运算', '方程', '函数', '几何图形',

            ],

            'computer_science': [
                '数据', '算法', '程序', '变量', '循环',
                'data', 'algorithm', 'program', 'variable', 'loop'
            ]
        }

    def explore_prerequisites(
        self,
        concept: str,
        subject: str,
        depth: int = 0,
        max_depth: int = 3
    ) -> Dict[str, Any]:
        """
        递归探索前置知识

        这是借鉴 Math2Manim 的核心创新！
        构建反向知识树，确保学习路径完整
        """
        # 基础情况：检查是否是基础概念
        if self._is_foundation_concept(concept, subject):
            return {
                'concept': concept,
                'subject': subject,
                'depth': depth,
                'prerequisites': [],
                'is_foundation': True
            }

        # 基础情况：达到最大深度
        if depth >= max_depth:
            return {
                'concept': concept,
                'subject': subject,
                'depth': depth,
                'prerequisites': [],
                'is_foundation': False,
                'note': '达到最大探索深度'
            }

        # 递归情况：查找前置知识
        prereqs = self._find_prerequisites(concept, subject)

        # 对每个前置知识递归探索
        prereq_trees = []
        for prereq in prereqs:
            prereq_tree = self.explore_prerequisites(
                prereq,
                subject,
                depth + 1,
                max_depth
            )
            prereq_trees.append(prereq_tree)

        return {
            'concept': concept,
            'subject': subject,
            'depth': depth,
            'prerequisites': prereq_trees,
            'is_foundation': False
        }

    def _is_foundation_concept(self, concept: str, subject: str) -> bool:
        """检查是否是基础概念"""
        foundation_list = self.foundation_concepts.get(subject, [])

        # 简单匹配
        for foundation in foundation_list:
            if foundation.lower() in concept.lower():
                return True

        # 使用 LLM 判断
        prompt = f"""
        概念: {concept}
        学科: {subject}

        判断这个概念是否是该学科的基础概念（高中水平或更低）。

        基础概念是指：
        - 不需要其他前置知识就能理解
        - 是该学科最基本的构建块
        - 高中生应该已经掌握

        只回答 "是" 或 "否"。
        """

        try:
            response = self.llm_client.generate(
                prompt,
                max_tokens=10,
                temperature=0
            )

            return '是' in response or 'yes' in response.lower()

        except:
            return False

    def _find_prerequisites(self, concept: str, subject: str) -> List[str]:
        """
        使用 LLM 查找前置知识

        问题: "要理解这个概念，必须先掌握什么？"
        """
        prompt = f"""
        学科: {subject}
        概念: {concept}

        问题: 要理解这个概念，学生必须先掌握哪些前置知识？

        要求：
        1. 列出 3-5 个最关键的前置概念
        2. 按重要性排序
        3. 只返回概念名称，用逗号分隔
        4. 不要包含解释
        5. 确保这些概念比当前概念更基础

        示例格式：
        前置概念1, 前置概念2, 前置概念3, 前置概念4, 前置概念5
        """

        try:
            response = self.llm_client.generate(
                prompt,
                max_tokens=200,
                temperature=0.3
            )

            # 解析响应
            prereqs = [p.strip() for p in response.split(',')]
            prereqs = [p for p in prereqs if len(p) > 1]  # 过滤掉太短的

            return prereqs[:5]  # 限制最多 5 个

        except Exception as e:
            print(f"查找前置知识失败: {e}")
            return []

    def visualize_tree(self, knowledge_tree: Dict[str, Any]) -> str:
        """可视化知识树（返回 Mermaid 格式）"""
        lines = ["graph TD"]

        def add_node(tree, parent_id=None):
            node_id = f"node_{id(tree)}"
            label = tree['concept'].replace('"', '\\"')

            lines.append(f'    {node_id}["{label}"]')

            if parent_id:
                lines.append(f'    {parent_id} --> {node_id}')

            for prereq in tree.get('prerequisites', []):
                add_node(prereq, node_id)

        add_node(knowledge_tree)
        return '\n'.join(lines)
```

```python
        match = re.search(pattern, response, re.DOTALL)
```

```python
# utils/llm_client.py

import os
from anthropic import Anthropic
from openai import OpenAI
from typing import Optional

class LLMClient:
    """统一的 LLM 客户端"""

    def __init__(self):
        self.anthropic_api_key = os.getenv("ANTHROPIC_API_KEY")
        self.openai_api_key = os.getenv("OPENAI_API_KEY")
        self.google_api_key = os.getenv("GOOGLE_API_KEY")

        # 初始化客户端
        if self.anthropic_api_key:
            self.anthropic_client = Anthropic(api_key=self.anthropic_api_key)

        if self.openai_api_key:
            self.openai_client = OpenAI(api_key=self.openai_api_key)

        # 默认使用 Claude
        self.default_provider = 'anthropic'
        self.default_model = 'claude-sonnet-4-5'

    def generate(
        self,
        prompt: str,
        system_message: Optional[str] = None,
        model: Optional[str] = None,
        max_tokens: int = 2000,
        temperature: float = 0.7
    ) -> str:
        """
        生成文本

        Args:
            prompt: 用户提示
            system_message: 系统消息
            model: 模型名称
            max_tokens: 最大 token 数
            temperature: 温度参数

        Returns:
            生成的文本
        """
        # 确定使用的模型
        if model is None:
            model = self.default_model

        # 根据模型选择提供商
        if 'claude' in model.lower() or 'anthropic' in model.lower():
            return self._generate_anthropic(prompt, system_message, model, max_tokens, temperature)
        elif 'gpt' in model.lower() or 'openai' in model.lower():
            return self._generate_openai(prompt, system_message, model, max_tokens, temperature)
        else:
            # 默认使用 Anthropic
            return self._generate_anthropic(prompt, system_message, model, max_tokens, temperature)

    def _generate_anthropic(
        self,
        prompt: str,
        system_message: Optional[str],
        model: str,
        max_tokens: int,
        temperature: float
    ) -> str:
        """使用 Anthropic API 生成"""
        if not self.anthropic_api_key:
            raise ValueError("ANTHROPIC_API_KEY not set")

        messages = [{"role": "user", "content": prompt}]

        try:
            response = self.anthropic_client.messages.create(
                model=model,
                max_tokens=max_tokens,
                temperature=temperature,
                system=system_message or "你是一个专业的AI助手。",
                messages=messages
            )

            return response.content[0].text

        except Exception as e:
            print(f"Anthropic API 调用失败: {e}")
            raise

    def _generate_openai(
        self,
        prompt: str,
        system_message: Optional[str],
        model: str,
        max_tokens: int,
        temperature: float
    ) -> str:
        """使用 OpenAI API 生成"""
        if not self.openai_api_key:
            raise ValueError("OPENAI_API_KEY not set")

        messages = []

        if system_message:
            messages.append({"role": "system", "content": system_message})

        messages.append({"role": "user", "content": prompt})

        try:
            response = self.openai_client.chat.completions.create(
                model=model,
                messages=messages,
                max_tokens=max_tokens,
                temperature=temperature
            )

            return response.choices[0].message.content

        except Exception as e:
            print(f"OpenAI API 调用失败: {e}")
            raise
```

```javascript
    constructor() {
        this.container = document.getElementById('analysisResults');
    }

    showLoading(message = '正在分析...') {
        this.container.innerHTML = `
            <div class="loading-spinner">
                <i class="fas fa-spinner fa-spin"></i>
                <p>${message}</p>
            </div>
        `;
        this.container.style.display = 'block';
    }

    showError(message) {
        this.container.innerHTML = `
            <div class="error-message">
                <i class="fas fa-exclamation-circle"></i>
                <p>${message}</p>
            </div>
        `;
    }

    displayResults(result) {
        let html = '';

        // 1. 文件信息
        html += this._renderFileInfo(result.file_info);

        // 2. 学科识别
        html += this._renderSubjectInfo(result.subject_info);

        // 3. 内容摘要
        html += this._renderContentSummary(result.content_summary);

        // 4. 关键词
        html += this._renderKeywords(result.keywords);

        // 5. 概念列表
        html += this._renderConcepts(result.concepts);

        // 6. 知识树
        if (result.knowledge_tree) {
            html += this._renderKnowledgeTree(result.knowledge_tree);
        }

        // 7. 可视化结果
        if (result.visualization) {
            html += this._renderVisualization(result.visualization);
        }

        this.container.innerHTML = html;
        this.container.style.display = 'block';

        // 渲染知识树图表
        if (result.knowledge_tree) {
            this._renderTreeChart(result.knowledge_tree);
        }

        // 渲染可视化
        if (result.visualization) {
            this._executeVisualization(result.visualization);
        }
    }

    _renderFileInfo(fileInfo) {
        return `
            <div class="result-section">
                <h4><i class="fas fa-file"></i> 文件信息</h4>
                <div class="info-grid">
                    <div class="info-item">
                        <span class="label">文件名</span>
                        <span class="value">${fileInfo.filename}</span>
                    </div>
                    <div class="info-item">
                        <span class="label">类型</span>
                        <span class="value">${fileInfo.type}</span>
                    </div>
                    <div class="info-item">
                        <span class="label">大小</span>
                        <span class="value">${this._formatFileSize(fileInfo.size)}</span>
                    </div>
                </div>
            </div>
        `;
    }

    _renderSubjectInfo(subjectInfo) {
        const confidenceClass = subjectInfo.confidence;
        const confidenceText = {
            'high': '高',
            'medium': '中',
            'low': '低'
        }[subjectInfo.confidence] || subjectInfo.confidence;

        return `
            <div class="result-section">
                <h4><i class="fas fa-graduation-cap"></i> 学科识别</h4>
                <div class="subject-badge primary">
                    ${this._translateSubject(subjectInfo.primary_subject)}
                    <span class="confidence ${confidenceClass}">
                        置信度: ${confidenceText}
                    </span>
                </div>
                <div class="matched-domains">
                    ${subjectInfo.matched_domains.map(d =>
                        `<span class="domain-tag">${this._translateDomain(d)}</span>`
                    ).join('')}
                </div>
            </div>
        `;
    }

    _renderKnowledgeTree(tree) {
        return `
            <div class="result-section">
                <h4><i class="fas fa-sitemap"></i> 知识结构树</h4>
                <div id="knowledgeTreeChart" class="tree-chart"></div>
            </div>
        `;
    }

    _renderTreeChart(tree) {
        // 使用 D3.js 或 Mermaid.js 渲染知识树
        // 这里使用 Mermaid.js
        const mermaidCode = this._generateMermaidCode(tree);

        const chartDiv = document.getElementById('knowledgeTreeChart');
        chartDiv.innerHTML = `<pre class="mermaid">${mermaidCode}</pre>`;

        // 触发 Mermaid 渲染
        if (window.mermaid) {
            mermaid.init();
        }
    }

    _generateMermaidCode(tree) {
        let code = 'graph TD\n';

        const nodes = [];
        const edges = [];

        const traverse = (node, parentId = null) => {
            const nodeId = `node_${nodes.length}`;
            const label = node.concept.replace(/"/g, '\\"');

            nodes.push({
                id: nodeId,
                label: label,
                isFoundation: node.is_foundation
            });

            if (parentId) {
                edges.push(`${parentId} --> ${nodeId}`);
            }

            if (node.prerequisites) {
                node.prerequisites.forEach(prereq => {
                    traverse(prereq, nodeId);
                });
            }
        };

        traverse(tree);

        // 生成节点定义
        nodes.forEach(node => {
            const style = node.isFoundation ? 'style:' + node.id + ' fill:#90EE90;' : '';
            code += `    ${node.id}["${node.label}"]\n`;
        });

        // 生成边定义
        edges.forEach(edge => {
            code += `    ${edge}\n`;
        });

        return code;
    }

    _renderVisualization(viz) {
        return `
            <div class="result-section">
                <h4><i class="fas fa-chart-bar"></i> 可视化结果</h4>
                <div class="visualization-container">
                    <div id="vizOutput"></div>
                </div>
                <details class="code-viewer">
                    <summary>查看生成的代码</summary>
                    <pre><code>${this._escapeHtml(viz.code)}</code></pre>
                </details>
            </div>
        `;
    }

    _executeVisualization(viz) {
        const outputDiv = document.getElementById('vizOutput');

        if (viz.engine === 'manim') {
            // 显示视频
            if (viz.output && viz.output.success) {
                outputDiv.innerHTML = `
                    <video controls style="max-width: 100%;">
                        <source src="/api/v1/files/${encodeURIComponent(viz.output.video_url)}" type="video/mp4">
                    </video>
                `;
            } else {
                outputDiv.innerHTML = `<p class="error">渲染失败: ${viz.output?.error || '未知错误'}</p>`;
            }
        } else if (viz.engine === 'd3.js') {
            // 在 iframe 中执行 D3.js 代码
            this._executeInIframe(viz.code, ['https://d3js.org/d3.v7.min.js'], outputDiv);
        } else if (viz.engine === 'three.js') {
            // 在 iframe 中执行 Three.js 代码
            this._executeInIframe(viz.code, ['https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js'], outputDiv);
        }
    }

    _executeInIframe(code, scriptUrls, container) {
        const iframe = document.createElement('iframe');
        iframe.style.width = '100%';
        iframe.style.height = '500px';
        iframe.style.border = 'none';

        const scriptTags = scriptUrls.map(url => `<script src="${url}"></script>`).join('\n');

        const html = `
            <!DOCTYPE html>
            <html>
            <head>
                ${scriptTags}
                <style>
                    body { margin: 0; overflow: hidden; }
                    #viz { width: 100%; height: 100%; }
                </style>
            </head>
            <body>
                <div id="viz"></div>
                <script>
                    ${code}
                </script>
            </body>
            </html>
        `;

        container.appendChild(iframe);
        iframe.srcdoc = html;
    }

    _formatFileSize(bytes) {
        if (bytes < 1024) return bytes + ' B';
        if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
        return (bytes / (1024 * 1024)).toFixed(1) + ' MB';
    }

    _translateSubject(subject) {
        const translations = {
            'mathematics': '数学',
            'physics': '物理',
            'chemistry': '化学',
            'biology': '生物',
            'computer_science': '计算机科学',
            'economics': '经济学',
            'general': '通用'
        };
        return translations[subject] || subject;
    }

    _translateDomain(domain) {
        const translations = {
            'calculus': '微积分',
            'geometry': '几何',
            'algebra': '代数',
            'mechanics': '力学',
            'quantum': '量子物理',
            'algorithms': '算法',
            'machine_learning': '机器学习'
        };
        return translations[domain] || domain;
    }

    _escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }
}
```

### 5.3 主应用逻辑

```javascript
// frontend-v4/js/main.js

class App {
    constructor() {
        this.api = new VisualizationAPI();
        this.resultDisplay = new ResultDisplay();
        this.fileUploadHandler = new FileUploadHandler();
    }

    init() {
        // 初始化文件上传
        this.fileUploadHandler.init();

        // 监听文件上传事件
        this.fileUploadHandler.on('fileUploaded', async (file) => {
            await this.handleFileUpload(file);
        });
    }

    async handleFileUpload(file) {
        try {
            // 显示加载状态
            this.resultDisplay.showLoading('正在分析文档，这可能需要几分钟...');

            // 调用 API
            const result = await this.api.analyzeDocument(file, {
                maxDepth: 3,
                language: 'zh-CN'
            });

            // 显示结果
            this.resultDisplay.displayResults(result);

        } catch (error) {
            console.error('Analysis failed:', error);
            this.resultDisplay.showError(`分析失败: ${error.message}`);
        }
    }
}

document.addEventListener('DOMContentLoaded', () => {
```
