### 勾股定理动画技术要点

```python
# 1. 正确的三角形顶点
triangle = Polygon(
    ORIGIN,        # (0,0) 直角顶点
    RIGHT * a,     # (a,0) 底边
    UP * b,        # (0,b) 高
    ...
)

# 2. 正方形对齐
square_a.next_to(triangle, DOWN, buff=0)
square_a.align_to(triangle, LEFT)

square_b.next_to(triangle, LEFT, buff=0)
square_b.align_to(triangle, DOWN)

# 3. 斜边正方形旋转对齐
hypotenuse_vector = np.array([a, -b, 0])
hypotenuse_unit_vector = hypotenuse_vector / np.linalg.norm(hypotenuse_vector)
perpendicular_vector = np.array([
    hypotenuse_unit_vector[1],
    -hypotenuse_unit_vector[0],
    0
])

angle = np.arctan2(-b, a)
square_c.rotate(angle - PI/4)
square_c.move_to(midpoint + perpendicular_vector * c * 0.7)

# 4. 颜色编码公式
formula = MathTex("a", "^2", "+", "b", "^2", "=", "c", "^2")
formula[0].set_color(YELLOW)  # a
formula[1].set_color(RED)     # a²
formula[7].set_color(GREEN)   # c²
```
